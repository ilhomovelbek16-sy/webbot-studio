    <script>
        // Narxlarni aniqlash lug'ati
        const prices = {
            "Telegram Bot (Oddiy) - $50": 50,
            "Telegram Bot (Internet do'kon) - $120": 120,
            "Landing Page - $80": 80,
            "Korporativ Veb-sayt - $200": 200
        };

        function calculateTotal() {
            const val = document.getElementById('projectType').value;
            const price = prices[val] || 50;
            document.getElementById('totalPrice').innerText = '$' + price;
        }

        async function sendOrder() {
            const name = document.getElementById('clientName').value.trim();
            const phone = document.getElementById('clientPhone').value.trim();
            const projectType = document.getElementById('projectType').value;
            const totalPrice = document.getElementById('totalPrice').innerText;
            const statusMsg = document.getElementById('statusMsg');

            if (!name || !phone) {
                statusMsg.style.color = '#f87171';
                statusMsg.innerText = 'Iltimos, ism va telefon raqamingizni kiriting!';
                return;
            }

            statusMsg.style.color = '#94a3b8';
            statusMsg.innerText = 'Buyurtma yuborilmoqda...';

            try {
                // TO'G'RILANDI: Internetda ishlashi uchun nisbiy manzilga o'zgartirildi
                const response = await fetch('/api/order', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ name, phone, project_type: projectType, total_price: totalPrice })
                });

                const result = await response.json();
                if (response.ok) {
                    statusMsg.style.color = '#4ade80';
                    statusMsg.innerText = 'Rahmat! Buyurtmangiz qabul qilindi, tez orada bog‘lanamiz.';
                    document.getElementById('clientName').value = '';
                    document.getElementById('clientPhone').value = '';
                } else {
                    throw new Error(result.detail || 'Xatolik yuz berdi');
                }
            } catch (error) {
                statusMsg.style.color = '#f87171';
                statusMsg.innerText = 'Xatolik: Server bilan aloqa yo\'q yoki backend ishga tushmagan.';
            }
        }
    </script>
